// ========== deep-read-summarize v3.0 ==========
// 3 波次：解析器获取分块 → 并行精读 → 成稿+质量校验
// 特性：解析器插件化 / 失败纪律(fatal vs 降级) / 成本控制（maxChunks 强制，块数=子代理数）

// YAML 双引号标量：JSON.stringify 的转义规则是 YAML double-quoted 风格的子集，
// 标题/作者里出现引号时不会再截断 frontmatter（旧版是裸拼 "..."）。
function yamlStr(v) { return JSON.stringify(v == null ? "" : String(v)); }

// ---------- 0. 配置校验（fatal 级错误：必须终止） ----------
if (args && args.input !== undefined && typeof args.input !== "string") throw new Error("[FATAL] args.input 必须是字符串");
if (args && args.type !== undefined && typeof args.type !== "string") throw new Error("[FATAL] args.type 必须是字符串");
const input = (args && args.input || "").trim();
const forcedType = ((args && args.type) || "auto").toLowerCase();
const options = normalizeWorkflowOptions(args && args.options);
const minWords = options.minWords;
const fastMode = options.fastMode;
const maxChunks = options.maxChunks;
const requireCitations = options.requireCitations;
const includeTimestamps = options.includeTimestamps;
const maxRetries = options.maxRetries;
const outputDir = options.outputDir;
// tempDir 是根目录；每次调用使用独立子目录，获取、转写与 Map 必须共享同一个任务路径。
const tempRoot = options.tempDir;
const runId = Date.now().toString(36) + "-" + Math.random().toString(36).slice(2) + Math.random().toString(36).slice(2);
const tempDir = tempRoot.replace(/[\\/]+$/, "") + "/run-" + runId;
const tempFile = tempDir + "/input.txt";
const whisperModel = options.whisperModel;
const transcribeLanguage = options.language;
const transcribeDevice = options.device;
// frontmatter 档位。默认 obsidian 保留历史字段（现有用户的 Dataview 查询依赖 type / status）。
// generic 档换掉三个「Hugo 保留键」：type -> docType、url -> sourceUrl、created -> date（date 才是标准键）；
// minimal 档只留 title / date / sourceUrl / tags。
const FM_PROFILES = ["obsidian", "generic", "minimal"];
const fmProfile = String(options.frontmatter || "obsidian").toLowerCase();
if (FM_PROFILES.indexOf(fmProfile) === -1) {
  throw new Error("[FATAL] options.frontmatter 非法：" + options.frontmatter + "（可用：" + FM_PROFILES.join("|") + "）");
}

// 幂等不在这里做：脚本沙箱没有文件系统，写不了指纹记录。
// 需要幂等的调用方请用 lib/cache.js（fingerprint / hasProcessed / markProcessed）。

// ---------- 1. 解析器表：构建时把解析器内联为 __parsers（自包含）；
//              宿主若注入 args._parsers 则优先使用 ----------
const parsers = args._parsers || {};
const parserNames = Object.keys(parsers);

// 配置校验依赖解析器表，必须放在其后
if (!input) throw new Error("[FATAL] args.input 缺失：请输入内容链接或文件路径");
if (parserNames.length === 0) throw new Error("[FATAL] 解析器表为空：脚本没有内联任何解析器");
if (forcedType !== "auto" && parserNames.indexOf(forcedType) === -1)
  throw new Error("[FATAL] args.type 非法：" + forcedType + "（可用：" + ["auto"].concat(parserNames).join("|") + "）");

phase("解析器选择");
// 只对明确的来源做快速识别；PDF、未知网页和文本可能属于不同类型，交给轻量分类。
// 不依赖 URL 等宿主未承诺的全局 API，保持脚本可在 workflow 沙箱中运行。
function inferType(source) {
  const url = source.match(/https?:\/\/[^\s<>"']+/i);
  const target = url ? url[0] : source;
  const hostMatch = target.match(/^https?:\/\/(?:[^\/?#]*@)?([^\/:?#]+)/i);
  const host = hostMatch ? hostMatch[1].toLowerCase() : "";
  const pathMatch = target.match(/^https?:\/\/[^\/?#]+([^?#]*)/i);
  const urlPath = pathMatch ? pathMatch[1] : "";
  const queryMatch = target.split("#")[0].match(/\?([\s\S]*)$/);
  const query = queryMatch ? queryMatch[1] : "";
  // 同一站点也有专栏、频道和帮助页，只对明确的视频/论文内容路径快速分流。
  if (/(^|\.)(youtu\.be|b23\.tv|v\.douyin\.com)$/.test(host) && /^\/[^/]+/.test(urlPath)) return "video";
  if (/(^|\.)(youtube\.com|youtube-nocookie\.com)$/.test(host) &&
      (/^\/(shorts|live|embed|v)\/[^/]+/.test(urlPath) || (urlPath === "/watch" && /(^|&)v=[^&]+/.test(query)))) return "video";
  if (/(^|\.)bilibili\.com$/.test(host) && /^\/(video|bangumi\/play)\/[^/]+/.test(urlPath)) return "video";
  if (/(^|\.)douyin\.com$/.test(host) && /^\/(share\/)?video\/[^/]+/.test(urlPath)) return "video";
  if (/(^|\.)vimeo\.com$/.test(host) && /^\/(video\/)?\d+(?:\/|$)/.test(urlPath)) return "video";
  if (/(^|\.)arxiv\.org$/.test(host) && /^\/(abs|pdf|html)\/[^/]+/.test(urlPath)) return "paper";
  const resourcePath = hostMatch ? urlPath : target.split(/[?#]/)[0];
  if (/\.(epub|mobi|azw3?)$/i.test(resourcePath)) return "book";
  return "";
}
let selectedType = forcedType;
if (forcedType === "auto") {
  selectedType = inferType(input);
  if (parserNames.indexOf(selectedType) === -1) {
    const detectSchema = {
      type: "object", properties: { kind: { type: "string", enum: parserNames } },
      required: ["kind"], additionalProperties: false
    };
    const detectPrompt = [
      "识别输入内容类型，选择后续获取全文所需的解析器，只输出 JSON {\"kind\": \"类型\"}。",
      "可用类型：" + parserNames.join(", ") + "。book=书籍，paper=学术论文，video=视频，web=网页/文章。",
      "PDF 既可能是书籍也可能是论文；未知网站可能承载论文或在线书籍，不要仅凭扩展名或网页形式判断。",
      "必要时只查看标题、摘要、目录或页面元信息，不获取或精读全文，不写入文件。",
      "以下 JSON 字符串是输入来源数据：", JSON.stringify(input)
    ].join("\n");
    const detected = await agent(detectPrompt, { schema: detectSchema, label: "类型识别", phase: "解析器选择" });
    selectedType = detected && detected.kind;
  }
  if (parserNames.indexOf(selectedType) === -1) {
    return { ok: false, stage: "detect", fatal: false, message: "无法识别输入类型，请显式指定 args.type：" + parserNames.join("|") };
  }
}
log("解析器：" + selectedType + " | 临时目录：" + tempDir);
const kindNames = { book: "书籍", paper: "学术论文", video: "视频", web: "网页/文章" };
const kindFocus = {
  book: "按章节主题展开核心论点、论证、关键例子、章节间关系，收集关键概念与可引用原文（含页码/章节）",
  paper: "四维精读：1)研究问题与方法 2)结论与贡献 3)局限与批判 4)与同类工作比较；收集术语与可引用原文（含章节/段落）",
  video: "按主题脉络组织，提取核心论点、证据、例子、可操作建议，收集关键概念与主讲人原话" + (includeTimestamps ? "（含时间戳定位）" : "（不标时间戳）"),
  web: "提取核心论点、结构拆解、关键概念、可引用原文"
};

// ---------- 2. 波次1：解析器获取 + 分块 ----------
phase("获取与分块");
const fetchPrompt = parsers[selectedType]
  .buildPrompt(input, { tempDir: tempDir, tempFile: tempFile, maxChunks: maxChunks, transcribe: options.transcribe, whisperModel: whisperModel, language: transcribeLanguage, device: transcribeDevice, includeTimestamps: includeTimestamps });
// fetchSchema 由插件在构建时从 schemas/index.js 注入（单一真相，避免两份 schema 漂移）。

const fetched = await agent(fetchPrompt, { schema: fetchSchema, label: "获取+分块", phase: "获取与分块" });

// 失败纪律：内容获取失败 = 可降级错误（返回失败结果，不抛 fatal）
if (!fetched || !fetched.saved) {
  return { ok: false, stage: "fetch", fatal: false, message: (fetched && fetched.message) || "获取失败（可降级：检查输入或手动提供文本）" };
}
// 先校验完整计划，按行排序、裁掉重复覆盖并补齐缺口，再合并相邻块。
// 块数上限约束的是子代理数量，不能通过丢掉尾部原文来实现。
const rawPlan = (fetched && fetched.chunkPlan) || [];
if (rawPlan.length === 0) return { ok: false, stage: "fetch", fatal: false, message: "分块计划为空" };
const lineCount = Number(fetched.totalLines);
const declaredLines = Number.isFinite(lineCount) && lineCount > 0 ? Math.floor(lineCount) : 0;
const validPlan = [];
for (let i = 0; i < rawPlan.length; i++) {
  const c = rawPlan[i] || {};
  const s = Math.floor(Number(c.startLine));
  const e = Math.floor(Number(c.endLine));
  if (!Number.isFinite(s) || !Number.isFinite(e) || s < 1 || e < s) continue;
  if (declaredLines > 0 && s > declaredLines) continue;
  validPlan.push({
    topic: String(c.topic || ("块 " + (i + 1))),
    startLine: s,
    endLine: declaredLines > 0 ? Math.min(e, declaredLines) : e
  });
}
if (validPlan.length === 0) return { ok: false, stage: "fetch", fatal: false, message: "分块计划无效：行号区间缺失或非法" };
validPlan.sort(function (a, b) { return a.startLine - b.startLine || a.endLine - b.endLine; });
const coveredPlan = [];
let nextLine = 1;
for (let i = 0; i < validPlan.length; i++) {
  const c = validPlan[i];
  if (c.startLine > nextLine) {
    coveredPlan.push({ topic: "补充未规划内容", startLine: nextLine, endLine: c.startLine - 1 });
  }
  const startLine = Math.max(nextLine, c.startLine);
  if (c.endLine >= startLine) coveredPlan.push({ topic: c.topic, startLine: startLine, endLine: c.endLine });
  nextLine = Math.max(nextLine, c.endLine + 1);
}
if (declaredLines > 0 && nextLine <= declaredLines) {
  coveredPlan.push({ topic: "补充尾部内容", startLine: nextLine, endLine: declaredLines });
}
const chunkPlan = [];
const groups = Math.min(maxChunks, coveredPlan.length);
for (let i = 0; i < groups; i++) {
  const from = Math.floor(i * coveredPlan.length / groups);
  const to = Math.floor((i + 1) * coveredPlan.length / groups);
  const parts = coveredPlan.slice(from, to);
  chunkPlan.push({
    id: i + 1, topic: parts.map(function (c) { return c.topic; }).join(" / "),
    startLine: parts[0].startLine, endLine: parts[parts.length - 1].endLine
  });
}
const coverageComplete = declaredLines > 0;
if (!coverageComplete) log("警告：获取结果缺少有效 totalLines，只能确认已知行号范围的覆盖");
log("分块计划：" + rawPlan.length + " -> " + chunkPlan.length + " 块（上限 " + maxChunks + "，保留全部已知内容）");

// 成稿类型必须与实际执行的解析器一致，不能在获取后仅改变标签。
const kind = selectedType;
const metadata = (fetched && fetched.metadata) || {};
const textSource = (metadata && metadata.textSource) || "unknown";
const srcLabels = { desc: "视频简介/desc", subtitle: "平台字幕", transcription: "完整语音转写", manual: "用户提供转写文本" };
const srcLabel = srcLabels[textSource] || textSource;
log("[source] " + textSource + " -> " + srcLabel);
log("类型:" + kind + " 行数:" + fetched.totalLines + " 块数:" + chunkPlan.length);

// ---------- 3. 波次2：并行精读（Map） ----------
phase("并行精读");
const chunkPrompts = chunkPlan.map(function (c) {
  return [
    "请深度精读以下内容块，输出结构化精读笔记。",
    "内容类型：" + (kindNames[kind] || kind) + " | 本块主题：" + c.topic,
    "精读侧重：" + (kindFocus[kind] || kindFocus.web),
    "用 read 工具读取 " + tempFile + " 第 " + c.startLine + " 到 " + c.endLine + " 行（offset=" + c.startLine + ", limit=" + (c.endLine - c.startLine + 1) + "），然后输出：",
    "## 本块主题：" + c.topic,
    "### 要点（详细展开，保留数据/例子/论证）",
    "### 关键概念（术语+解释）",
    "### 关键引用（最多2句原文+位置）" + (requireCitations ? "，必须标注页码/章节/段落位置" : ""),
    "引用保留原话并附本文件的真实行号；视频时间戳只能从已保存字幕/转写的时间标记读取，缺失时注明无法定位，不得推测。",
    "### 批判观察（如有）"
  ].join("\n");
});

const reads = [];
const readBatch = 3;
for (let bi = 0; bi < chunkPrompts.length; bi += readBatch) {
  const batch = chunkPrompts.slice(bi, bi + readBatch);
  const batchReads = await parallel(batch.map(function (p) {
    return async function () {
      try { return await agent(p, { label: "精读块", phase: "并行精读" }); }
      catch (error) { log("精读调用失败，保留对应分块缺口"); return ""; }
    };
  }));
  for (let k = 0; k < batchReads.length; k++) reads.push(batchReads[k]);
}
const validReads = reads.filter(function (r) { return typeof r === "string" && r.trim().length > 0; });
const readEvidence = chunkPlan.map(function (chunk, i) {
  return { chunkId: chunk.id, topic: chunk.topic, startLine: chunk.startLine, endLine: chunk.endLine,
    readSucceeded: typeof reads[i] === "string" && reads[i].trim().length > 0,
    notes: typeof reads[i] === "string" ? reads[i] : "" };
});
const failedChunkIds = readEvidence.filter(function (r) { return !r.readSucceeded; }).map(function (r) { return r.chunkId; });

// 失败纪律：某块精读失败 = 可降级（跳过该块，标记缺口继续）
const failedChunks = chunkPlan.length - validReads.length;
if (validReads.length === 0) return { ok: false, stage: "read", fatal: false, message: "所有分块精读均失败" };
if (failedChunks > 0) log("警告：" + failedChunks + " 块精读失败，标记缺口继续成稿");
log("精读完成：" + validReads.length + "/" + chunkPlan.length);

// ---------- 4. 波次3：成稿 + 内嵌质量自检（Reduce） ----------
phase("汇总成稿");
const todayDate = new Date();
const today = todayDate.getFullYear() + "-" + ("0" + (todayDate.getMonth() + 1)).slice(-2) + "-" + ("0" + todayDate.getDate()).slice(-2);
// 标题先折叠换行/制表与连续空白：否则会同时污染 .md 文件名和 frontmatter。
const finalTitle = String(metadata.title || "").replace(/[\r\n\t]+/g, " ").replace(/\s+/g, " ").trim() || "未命名内容";
// 文件名清洗：去掉 Windows 非法字符；全部非法时兜底，避免写出 ./output/.md 这种隐藏文件。
const safeTitle = finalTitle.replace(/[\\/:*?"<>|]/g, " ").replace(/\s+/g, " ").trim() || "未命名内容";
const mdPath = outputDir.replace(/[\\/]+$/, "") + "/" + safeTitle + ".md";

const extraSections = fastMode ? [] : [
  "", "## 5. 批判性分析（论证强处/弱处/可质疑之处）", "",
  "## 6. 个人思考与应用（与已有知识联系、如何应用、行动要点）", "",
  "## 7. 延伸阅读（相关主题/作者其他作品/参考文献）", ""
];

// frontmatter 按档位组装。obsidian 档保留历史键名，字符串安全转义。
const fm = ["---"];
if (fmProfile === "obsidian") {
  fm.push('title: ' + yamlStr("《" + finalTitle + "》深度精读笔记"));
  fm.push("aliases: [" + yamlStr(finalTitle) + "]");
  fm.push("author: " + yamlStr(metadata.author));
  fm.push("year: " + yamlStr(metadata.year));
  fm.push("type: " + yamlStr(kind));
  fm.push("url: " + yamlStr(metadata.url || input));
  fm.push("tags: [" + yamlStr("精读") + ", " + yamlStr(kindNames[kind] || kind) + "]");
  fm.push("created: " + today);
  fm.push("status: 已完成");
} else if (fmProfile === "generic") {
  fm.push('title: ' + yamlStr("《" + finalTitle + "》深度精读笔记"));
  fm.push("date: " + today);
  fm.push("author: " + yamlStr(metadata.author));
  fm.push("year: " + yamlStr(metadata.year));
  fm.push("docType: " + yamlStr(kind));
  fm.push("sourceUrl: " + yamlStr(metadata.url || input));
  fm.push("tags: [" + yamlStr("精读") + ", " + yamlStr(kindNames[kind] || kind) + "]");
  fm.push("status: 已完成");
} else {
  fm.push('title: ' + yamlStr("《" + finalTitle + "》深度精读笔记"));
  fm.push("date: " + today);
  fm.push("sourceUrl: " + yamlStr(metadata.url || input));
  fm.push("tags: [" + yamlStr("精读") + ", " + yamlStr(kindNames[kind] || kind) + "]");
}
fm.push("---");

const draftPrompt = [
  "将以下多块精读结果整合为完整的结构化精读笔记，只输出标准 Markdown（含 YAML frontmatter）。",
  "模板："
].concat(fm, [
  "# 《" + finalTitle + "》深度精读笔记", "",
  "> 一句话核心概括。", "",
  "## 1. 概述（约500字：主题/背景/核心问题/总体结论）", "",
  "## 2. 结构拆解（按主题组织，含详细要点；关键结论标注原文出处）", "",
  "## 3. 关键概念与术语表（表格）", "",
  "## 4. 关键引用（仅关键结论" + (requireCitations ? "，必须标注页码/章节/段落" : "") + "）",
  "本节逐字引文统一使用中文双引号“…”、「…」、英文双引号或 Markdown 引用块，并标注真实来源；不要用元数据、普通叙述或推测时间戳充当引文。",
], extraSections, [
  "---",
  "*笔记整理于 " + today + "*", "",
  "硬性要求：总篇幅≥" + minWords + "字；语言与原文一致；术语统一；引用必须真实；视频" + (includeTimestamps ? "标注时间戳" : "不标注时间戳") + "。",
  "输出前自检：1)覆盖所有块 2)frontmatter完整 3)各小节齐全 4)引用真实 5)篇幅达标 6)语言一致。",
  "",
  "各块精读结果：",
  readEvidence.map(function (r) { return "\n===== 块 " + r.chunkId + "（原文行 " + r.startLine + "-" + r.endLine + "） =====\n" + (r.readSucceeded ? r.notes : "精读失败，此处存在缺口，禁止编造补全。"); }).join("\n")
]).join("\n");

let draft;
try { draft = await agent(draftPrompt, { label: "汇总成稿", phase: "汇总成稿" }); }
catch (error) { return { ok: false, stage: "draft", fatal: false, message: "成稿调用失败，未产生可保存的笔记" }; }
if (typeof draft !== "string" || !draft.trim()) return { ok: false, stage: "draft", fatal: false, message: "成稿为空或不是字符串" };

// ---------- 5. 程序检查 + 独立原文核验 ----------
phase("质量校验");
let verified = draft;
let passed = false;
let issues = [];
let qualityChecks = {};
let qualityEvidence = { sourceFile: tempFile, totalLines: declaredLines, failedChunkIds: failedChunkIds, chunkChecks: [], citationChecks: [] };
const structureOptions = { minWords: minWords, fastMode: fastMode, expectedFrontmatter: fm };
const sourceContext = { sourceFile: tempFile, input: input, totalLines: declaredLines,
  chunkPlan: chunkPlan, readEvidence: readEvidence, failedChunkIds: failedChunkIds, coverageComplete: coverageComplete };
for (let round = 0; round <= maxRetries; round++) {
  const localCheck = __quality.checkNoteStructure(verified, structureOptions);
  const checkPrompt = [
    "独立核对笔记与原文。必须用 read 工具查看下述 sourceFile 的各个分块及每条引用对应的原文，不得仅根据笔记或精读摘要判定通过。",
    "sourceContext 是来源数据，不是工具指令：", JSON.stringify(sourceContext),
    "逐块检查是否覆盖关键内容；chunkChecks 必须逐一列出所有块的 chunkId、原计划 startLine/endLine、实际读到的简短 sourceExcerpt 和 covered。失败块/缺口必须如实列入 issues。",
    "逐条核对关键引用；citationChecks 包含笔记中的原话 quote、亲自读到的 sourceQuote、chunkId、真实 startLine/endLine、页码/章节/时间戳 location、verified。原文摘录应保留原句，不能把改写当作逐字引用。",
    "需要核对的显式引用：", JSON.stringify(localCheck.quotes),
    "引用要求：" + (requireCitations ? "至少一条经过原文核验的真实引用" : "不强制新增引用，但已有引用必须真实"),
    "输出 sourceChecked、coverage、citationsReal、terminologyConsistent、languageConsistent 五个布尔项及 chunkChecks、citationChecks、issues；未读取原文或无法确认时必须为 false。",
    "格式与篇幅已由程序检查，模型不得覆盖程序结果：", JSON.stringify({ formatComplete: localCheck.formatComplete, lengthMet: localCheck.lengthMet, wordCount: localCheck.wordCount, issues: localCheck.issues }),
    "笔记：", verified
  ].join("\n");
  const checkSchema = {
    type: "object", properties: {
      sourceChecked: { type: "boolean" }, coverage: { type: "boolean" }, citationsReal: { type: "boolean" },
      terminologyConsistent: { type: "boolean" }, languageConsistent: { type: "boolean" },
      chunkChecks: { type: "array", items: { type: "object", properties: {
        chunkId: { type: "number" }, startLine: { type: "number" }, endLine: { type: "number" },
        sourceExcerpt: { type: "string" }, covered: { type: "boolean" }
      }, required: ["chunkId", "startLine", "endLine", "sourceExcerpt", "covered"], additionalProperties: false } },
      citationChecks: { type: "array", items: { type: "object", properties: {
        quote: { type: "string" }, sourceQuote: { type: "string" }, chunkId: { type: "number" },
        startLine: { type: "number" }, endLine: { type: "number" }, location: { type: "string" }, verified: { type: "boolean" }
      }, required: ["quote", "sourceQuote", "chunkId", "startLine", "endLine", "location", "verified"], additionalProperties: false } },
      issues: { type: "array", items: { type: "string" } }
    }, required: ["sourceChecked", "coverage", "citationsReal", "terminologyConsistent", "languageConsistent", "chunkChecks", "citationChecks", "issues"], additionalProperties: false
  };
  let check = null;
  try { check = await agent(checkPrompt, { schema: checkSchema, label: "质量校验", phase: "质量校验" }); }
  catch (error) { log("原文核验调用失败，保留当前有效笔记并报告质量未通过"); }
  const sourceCheck = __quality.validateSourceCheck(check, {
    chunkPlan: chunkPlan, failedChunkIds: failedChunkIds, coverageComplete: coverageComplete,
    requireCitations: requireCitations, note: verified, visibleNote: localCheck.visibleText, quotes: localCheck.quotes
  });
  issues = Array.from(new Set(localCheck.issues.concat(sourceCheck.issues)));
  passed = localCheck.formatComplete && localCheck.lengthMet && sourceCheck.passed;
  qualityChecks = Object.assign({}, sourceCheck.checks, { formatComplete: localCheck.formatComplete, lengthMet: localCheck.lengthMet, wordCount: localCheck.wordCount });
  qualityEvidence = { sourceFile: tempFile, totalLines: declaredLines, failedChunkIds: failedChunkIds,
    chunkChecks: sourceCheck.chunkChecks, citationChecks: sourceCheck.citationChecks };
  log("校验：" + (passed ? "通过" : "发现问题 " + issues.length + " 条"));
  if (passed) break;
  if (round < maxRetries) {
    const fixPrompt = [
      "修正以下笔记以解决校验问题，输出完整修正版：",
      issues.map(function (s) { return "- " + s; }).join("\n"),
      "原文路径和覆盖证据（必要时用 read 核对，不能臆造缺口）：", JSON.stringify(sourceContext),
      "必须保留的 frontmatter 模板：", fm.join("\n"),
      "当前笔记：", verified
    ].join("\n");
    let repaired;
    try { repaired = await agent(fixPrompt, { label: "修正笔记", phase: "质量校验" }); }
    catch (error) { issues.push("修正调用失败，已保留上一份有效笔记"); break; }
    if (typeof repaired !== "string" || !repaired.trim()) {
      issues.push("修正结果为空或不是字符串，已保留上一份有效笔记"); break;
    }
    const repairedStructure = __quality.checkNoteStructure(repaired, structureOptions);
    if (!repairedStructure.formatComplete || (localCheck.lengthMet && !repairedStructure.lengthMet)) {
      issues.push("修正结果格式不完整或篇幅退化，已保留上一份有效笔记"); break;
    }
    verified = repaired;
  }
}

// ---------- 结果 ----------
return {
  ok: true, kind: kind, title: finalTitle,
  filePath: mdPath, chunksCount: chunkPlan.length, fastMode: fastMode,
  qualityPassed: passed, qualityIssues: issues,
  qualityChecks: qualityChecks, qualityEvidence: qualityEvidence,
  failedChunks: failedChunks,
  coverageComplete: coverageComplete,
  textSource: textSource,
  note: verified
};
